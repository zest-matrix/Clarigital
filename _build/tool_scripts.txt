<script>
(function(){
  var ham=document.getElementById('navHam'),drawer=document.getElementById('navDrawer'),close=document.getElementById('navClose');
  function open(){drawer.classList.add('open');ham.setAttribute('aria-expanded','true');document.body.style.overflow='hidden';}
  function shut(){drawer.classList.remove('open');ham.setAttribute('aria-expanded','false');document.body.style.overflow='';}
  if(ham)ham.addEventListener('click',open);
  if(close)close.addEventListener('click',shut);
  if(drawer)drawer.addEventListener('click',function(e){if(e.target===drawer)shut();});
})();
</script>
<script>
(function(){
  // Lane tabs
  var tabs=[];
  var sections=[];
  function showLane(color){
    tabs.forEach(function(t){t.classList.toggle('active',t.id==='tab-'+color);});
    sections.forEach(function(s){s.classList.toggle('active',s.id==='lane-'+color);});
  }
  window.showLane=showLane;
  // Default active tab
  var first=document.querySelector('.lane-tab');
  if(first){var m=first.id.match(/tab-(\w+)/);if(m)showLane(m[1]);}
  // Prompt copy
  window.copyPrompt=function(btn){
    var text=btn.previousElementSibling.textContent;
    navigator.clipboard.writeText(text).then(function(){
      var orig=btn.textContent;btn.textContent='Copied!';btn.classList.add('copied');
      setTimeout(function(){btn.textContent=orig;btn.classList.remove('copied');},1800);
    });
  };
})();
</script>
